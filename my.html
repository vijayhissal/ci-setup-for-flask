⚙️ STEP 2: Create Dockerfile
Writing

FROM python:3.9

WORKDIR /app

COPY . .

RUN pip install -r requirements.txt

CMD ["python", "app.py"]

⚙️ STEP 3: Build & Test Locally
docker build -t flask-ci-app .
docker run -d -p 5000:5000 --name flask-container flask-ci-app

⚙️ STEP 6: Add Build Commands

👉 Build → Execute Windows Batch Command

echo Removing old container
docker rm -f flask-container 2>nul

echo Running test
findstr "Flask" requirements.txt

IF %ERRORLEVEL% == 0 (
echo Test Passed ✅
) ELSE (
echo Test Failed ❌
exit /b 1
)

echo Building Docker image
docker build -t flask-ci-app .

echo Running container
docker run -d -p 5000:5000 --name flask-container flask-ci-app
